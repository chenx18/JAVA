```java
package com.example.admin.project.system.domain.request;

import jakarta.validation.constraints.Email;
import jakarta.validation.constraints.Max;
import jakarta.validation.constraints.Min;
import jakarta.validation.constraints.NotBlank;
import jakarta.validation.constraints.NotNull;
import jakarta.validation.constraints.Pattern;
import jakarta.validation.constraints.Size;
import lombok.Getter;
import lombok.Setter;

@Getter
@Setter
public class UserCreateRequest {
    @NotBlank(message = "用户名不能为空")
    @Pattern(regexp = "[A-Za-z0-9_]{3,50}", message = "用户名须为3到50位字母、数字或下划线")
    private String userName;

    @NotBlank(message = "密码不能为空")
    @Size(min = 8, max = 64, message = "密码须为8到64个字符")
    private String password;

    @NotBlank(message = "昵称不能为空")
    @Size(max = 50, message = "昵称不能超过50个字符")
    private String nickname;

    @Email(message = "邮箱格式不正确")
    @Size(max = 100, message = "邮箱不能超过100个字符")
    private String email;

    @NotNull(message = "状态不能为空")
    @Min(value = 0, message = "状态只能为0或1")
    @Max(value = 1, message = "状态只能为0或1")
    private Integer status = 1;
}
```

省略 status 使用默认值 1，显式传 null 则校验失败。邮箱可不填，`@Email` 不替代必填校验；需要必填时再配合 `@NotBlank`。

## 三、Mapper 与 XML

文件：`src/main/java/com/example/admin/project/system/mapper/UserMapper.java`

```java
package com.example.admin.project.system.mapper;

import com.example.admin.project.system.domain.User;
import java.util.List;
import org.apache.ibatis.annotations.Mapper;
import org.apache.ibatis.annotations.Param;

@Mapper
public interface UserMapper {
    User findById(@Param("id") Long id);
    User findByUserName(@Param("userName") String userName);
    long count(@Param("userName") String userName);
    List<User> findPage(@Param("userName") String userName,
                        @Param("offset") long offset,
                        @Param("pageSize") int pageSize);
    int insert(User user);
    int updateById(User user);
    int deleteById(@Param("id") Long id);
}
```

文件：`src/main/resources/mapper/system/UserMapper.xml`

```xml
<?xml version="1.0" encoding="UTF-8"?>
<!DOCTYPE mapper PUBLIC "-//mybatis.org//DTD Mapper 3.0//EN"
        "https://mybatis.org/dtd/mybatis-3-mapper.dtd">
<mapper namespace="com.example.admin.project.system.mapper.UserMapper">
  <resultMap id="UserResultMap" type="com.example.admin.project.system.domain.User">
    <id property="id" column="id"/>
    <result property="userName" column="user_name"/>
    <result property="password" column="password"/>
    <result property="nickname" column="nickname"/>
    <result property="email" column="email"/>
    <result property="status" column="status"/>
    <result property="createTime" column="create_time"/>
    <result property="updateTime" column="update_time"/>
  </resultMap>

  <sql id="PublicColumns">
    id, user_name, nickname, email, status, create_time, update_time
  </sql>

  <sql id="NameFilter">
    <where>
      <if test="userName != null and userName != ''">
        user_name like concat('%', replace(#{userName}, '_', '!_'), '%') escape '!'
      </if>
    </where>
  </sql>

  <select id="findById" resultMap="UserResultMap">
    select <include refid="PublicColumns"/> from sys_user where id = #{id}
  </select>

  <select id="findByUserName" resultMap="UserResultMap">
    select <include refid="PublicColumns"/>, password
    from sys_user where user_name = #{userName}
  </select>

  <select id="count" resultType="long">
    select count(*) from sys_user <include refid="NameFilter"/>
  </select>

  <select id="findPage" resultMap="UserResultMap">
    select <include refid="PublicColumns"/> from sys_user
    <include refid="NameFilter"/>
    order by id desc
    limit #{offset}, #{pageSize}
  </select>

  <insert id="insert" useGeneratedKeys="true" keyProperty="id">
    insert into sys_user (user_name, password, nickname, email, status)
    values (#{userName}, #{password}, #{nickname}, #{email}, #{status})
  </insert>

  <update id="updateById">
    update sys_user
    set user_name = #{userName}, nickname = #{nickname},
        email = #{email}, status = #{status}
    where id = #{id}
  </update>

  <delete id="deleteById">
    delete from sys_user where id = #{id}
  </delete>
</mapper>
```

`PublicColumns` 和 `NameFilter` 是可复用 SQL 片段，保证分页和总数使用相同过滤条件。筛选只允许字母、数字、下划线；下划线在 LIKE 中原本是通配符，这里转义后按字面量搜索。

`findByUserName` 供后续登录校验密码使用；普通详情和列表 SQL 不读取密码。即使实体读到了密码，也必须先转为 `UserResponse` 才能返回。

### 2. 完整 Service

文件：`src/main/java/com/example/admin/project/system/service/UserService.java`

```java
package com.example.admin.project.system.service;

import com.example.admin.common.exception.BusinessException;
import com.example.admin.common.response.PageResponse;
import com.example.admin.project.system.domain.User;
import com.example.admin.project.system.domain.request.UserCreateRequest;
import com.example.admin.project.system.domain.request.UserQueryRequest;
import com.example.admin.project.system.domain.request.UserUpdateRequest;
import com.example.admin.project.system.domain.vo.UserResponse;
import com.example.admin.project.system.mapper.UserMapper;
import java.nio.charset.StandardCharsets;
import java.util.List;
import lombok.RequiredArgsConstructor;
import org.springframework.dao.DuplicateKeyException;
import org.springframework.http.HttpStatus;
import org.springframework.security.crypto.password.PasswordEncoder;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

@Service
@RequiredArgsConstructor
public class UserService {
    private final UserMapper mapper;
    private final PasswordEncoder passwordEncoder;

    public UserResponse getUserById(Long id) {
        return toResponse(requireUser(id));
    }

    public PageResponse<UserResponse> getUsers(UserQueryRequest request) {
        long total = mapper.count(request.getUserName());
        long offset = (request.getPageNum() - 1L) * request.getPageSize();
        List<UserResponse> records = mapper.findPage(request.getUserName(),
                        offset, request.getPageSize())
                .stream().map(this::toResponse).toList();
        return new PageResponse<>(total, request.getPageNum(), request.getPageSize(), records);
    }

    @Transactional
    public UserResponse createUser(UserCreateRequest request) {
        checkUserName(request.getUserName(), null);
        // BCrypt 限制的是 UTF-8 字节数，中文字符可能占多个字节。
        if (request.getPassword().getBytes(StandardCharsets.UTF_8).length > 72) {
            throw new BusinessException(HttpStatus.BAD_REQUEST, "密码不能超过72个UTF-8字节");
        }
        User user = new User();
        user.setUserName(request.getUserName());
        user.setPassword(passwordEncoder.encode(request.getPassword()));
        user.setNickname(request.getNickname().trim());
        user.setEmail(normalizeEmail(request.getEmail()));
        user.setStatus(request.getStatus());
        try {
            mapper.insert(user);
        } catch (DuplicateKeyException e) {
            throw new BusinessException(HttpStatus.CONFLICT, "用户名已存在");
        }
        return toResponse(requireUser(user.getId()));
    }

    @Transactional
    public UserResponse updateUser(Long id, UserUpdateRequest request) {
        requireUser(id);
        checkUserName(request.getUserName(), id);
        User user = new User();
        user.setId(id);
        user.setUserName(request.getUserName());
        user.setNickname(request.getNickname().trim());
        user.setEmail(normalizeEmail(request.getEmail()));
        user.setStatus(request.getStatus());
        try {
            if (mapper.updateById(user) == 0) {
                throw new BusinessException(HttpStatus.NOT_FOUND, "用户不存在");
            }
        } catch (DuplicateKeyException e) {
            throw new BusinessException(HttpStatus.CONFLICT, "用户名已存在");
        }
        return toResponse(requireUser(id));
    }

    @Transactional
    public void deleteUser(Long id) {
        if (mapper.deleteById(id) == 0) {
            throw new BusinessException(HttpStatus.NOT_FOUND, "用户不存在");
        }
    }

    private User requireUser(Long id) {
        User user = mapper.findById(id);
        if (user == null) {
            throw new BusinessException(HttpStatus.NOT_FOUND, "用户不存在");
        }
        return user;
    }

    private void checkUserName(String userName, Long excludedId) {
        User existing = mapper.findByUserName(userName);
        if (existing != null && !existing.getId().equals(excludedId)) {
            throw new BusinessException(HttpStatus.CONFLICT, "用户名已存在");
        }
    }

    private String normalizeEmail(String email) {
        return email == null || email.isBlank() ? null : email.trim();
    }

    private UserResponse toResponse(User user) {
        return new UserResponse(user.getId(), user.getUserName(), user.getNickname(),
                user.getEmail(), user.getStatus(), user.getCreateTime(), user.getUpdateTime());
    }
}
```

先读懂一条新增链路：**校验 → 检查重名 → 哈希密码 → 插入 → 回填 ID → 查询最新记录 → 转换响应**。查询最新记录是为了返回数据库生成的时间等字段。

`@Transactional` 这里为写操作提供一致的边界，方便以后增加关联记录；单条 SQL 本身已具有原子性。方法要通过 Spring 注入的 Service 调用，类内自调用不会自动开启新的事务。

MySQL Connector/J 默认 `useAffectedRows=false`，UPDATE 返回匹配行数，内容没变的正常修改仍可成功。若改为 `true`，返回值语义发生变化，不能再只根据 0 判断不存在。本教程保持默认值。

查重后仍捕获 `DuplicateKeyException`，应对并发插入同名用户。本表此时只有用户名业务唯一键；以后增加邮箱等唯一键时，要重新区分冲突原因，不能一律返回“用户名已存在”。

分页的总数与列表是两次查询，并发写入时允许短暂差异。要求严格一致的报表需明确快照事务策略；此处普通管理列表不额外承诺快照一致性。

## 五、Controller

文件：`src/main/java/com/example/admin/project/system/controller/UserController.java`

```java
package com.example.admin.project.system.controller;

import com.example.admin.common.response.ApiResponse;
import com.example.admin.common.response.PageResponse;
import com.example.admin.project.system.domain.request.UserCreateRequest;
import com.example.admin.project.system.domain.request.UserQueryRequest;
import com.example.admin.project.system.domain.request.UserUpdateRequest;
import com.example.admin.project.system.domain.vo.UserResponse;
import com.example.admin.project.system.service.UserService;
import jakarta.validation.Valid;
import jakarta.validation.constraints.Positive;
import lombok.RequiredArgsConstructor;
import org.springdoc.core.annotations.ParameterObject;
import org.springframework.web.bind.annotation.DeleteMapping;
import org.springframework.web.bind.annotation.GetMapping;
import org.springframework.web.bind.annotation.ModelAttribute;
import org.springframework.web.bind.annotation.PathVariable;
import org.springframework.web.bind.annotation.PostMapping;
import org.springframework.web.bind.annotation.PutMapping;
import org.springframework.web.bind.annotation.RequestBody;
import org.springframework.web.bind.annotation.RequestMapping;
import org.springframework.web.bind.annotation.RestController;

@RestController
@RequestMapping("/system/users")
@RequiredArgsConstructor
public class UserController {
    private final UserService service;

    @GetMapping("/list")
    public ApiResponse<PageResponse<UserResponse>> getUsers(
            @Valid @ModelAttribute @ParameterObject UserQueryRequest request) {
        return ApiResponse.success(service.getUsers(request));
    }

    @GetMapping("/{id}")
    public ApiResponse<UserResponse> getUserById(@PathVariable @Positive Long id) {
        return ApiResponse.success(service.getUserById(id));
    }

    @PostMapping("/add")
    public ApiResponse<UserResponse> createUser(@Valid @RequestBody UserCreateRequest request) {
        return ApiResponse.success(service.createUser(request));
    }

    @PutMapping("/{id}")
    public ApiResponse<UserResponse> updateUser(@PathVariable @Positive Long id,
            @Valid @RequestBody UserUpdateRequest request) {
        return ApiResponse.success(service.updateUser(id, request));
    }

    @DeleteMapping("/{id}")
    public ApiResponse<Void> deleteUser(@PathVariable @Positive Long id) {
        service.deleteUser(id);
        return ApiResponse.success();
    }
}
```

GET 的查询对象用 `@ModelAttribute` 绑定 Query 参数，`@Valid` 一样可以执行校验。`@ParameterObject` 只是让 Swagger 展开字段，不改变 HTTP 传参方式。

本例使用 Spring MVC 自带的方法参数校验处理 `@Positive`，Controller 不加类级 `@Validated`，避免切换到另一套异常路径。

## 六、运行与 Apifox 验收

先执行 `./mvnw.cmd clean verify`。重新加载 `.env` 并启动应用，方式同上一节。不要把仅有健康测试通过误认为 CRUD 也通过，下面的真实数据库测试仍需执行。

在 Apifox 建立“本地开发”环境，设置 `baseUrl=http://localhost:8081`。按次序执行：

| 步骤 | 请求与操作 | 预期 |
| --- | --- | --- |
| 1 | POST `{{baseUrl}}/system/users/add`，Body/JSON 使用本节示例 | 200，记下真实 `data.id` |
| 2 | 完全重复新增 | HTTP 409，code 409 |
| 3 | GET `/system/users/{实际id}`，替换 Path 中的 ID | 200，无 password 字段 |
| 4 | GET `/system/users/list?pageNum=1&pageSize=10&userName=ali` | total 至少为 1，records 包含 alice |
| 5 | PUT `/system/users/{实际id}`，Body 为下方 JSON | 200，昵称和状态更新 |
| 6 | 再提交相同 PUT | 200，不能误判为用户不存在 |
| 7 | GET `/system/users/list?pageNum=0&pageSize=10` | HTTP 400 |
| 8 | GET `/system/users/list?pageSize=101` | HTTP 400 |
| 9 | POST 时传非法 email、空 userName 或 status=2 | HTTP 400，数据库不新增 |
| 10 | GET `/system/users/abc` 或 `/system/users/0` | HTTP 400 |
| 11 | PUT Body 只传空对象 `{}` | HTTP 400，必填资料缺失 |
| 12 | DELETE `/system/users/{实际id}` | 200，data 为 null |
| 13 | 再查询或删除刚才的 ID | HTTP 404 |

修改请求（没有 password）：

```json
{
  "userName": "alice",
  "nickname": "小林同学",
  "email": null,
  "status": 0
}
```

新增第二个用户验证：把它改名为 `alice` 应返回 409，而 `alice` 修改自己的昵称不会误报重名。查询没有匹配结果或页码超出最后一页时，`records` 是空数组，正常返回 200。

在本地数据库 UI 中核对 password 为 BCrypt 哈希，通常以 `$2` 开头，且不是请求明文。不要把真实密码或哈希贴到公开截图中。给登录练习准备一个启用的用户，记录自己的测试账号密码；示例没有预置 admin 账号。

## 七、知识沉淀

- **Mapper 的 int 是 SQL 影响/匹配行数。** Service 根据它决定业务结果，再把对象或 void 交给 Controller。
- **路径 ID 决定目标对象。** `user.setId(id)` 明确使用 URL 的 ID，不允许请求体选另一个用户。
- **数据库约束是并发兜底。** `@Valid` 管参数格式，Service 管规则，唯一索引负责最终一致的唯一性。
- **BCrypt 是带盐的密码哈希。** 登录时调用 `matches(明文, 已存哈希)`；重新 encode 后字符串比较不成立，也没有“解密密码”。
- **Request/Response 控制边界。** 本例已有证据需要分开：创建接密码，修改不接密码，查询永不返回密码。
- **基本 CRUD 不是完整账户系统。** 当前未实现授权、管理员保护和改密，必须在下一阶段补齐身份与权限后才能对外提供管理功能。

## 八、面试小题

1. **同名并发请求如何处理？** 提前查重改善提示，唯一键阻止竞争条件，异常转换为 HTTP 409。
2. **为什么更新后还查一次？** 返回数据库实际保存的值与时间；若接口不需要最新对象，也可以只返回成功状态。
3. **为什么 count 和列表共享过滤条件？** 避免页面总数与列表条件不同，造成分页错误。
4. **只写 @Transactional 就足够了吗？** 还要确认代理调用、异常传播、存储引擎以及操作是否属于同一数据库事务；Redis 不会自动随 MySQL 回滚。
